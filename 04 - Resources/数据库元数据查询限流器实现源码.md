## 一、连接池实时获取数据
```java
@Component
public class ConnectionPoolMonitor {

    private final HikariDataSource dataSource;
	
	//Springboot启动时自动根据yml创建连接池实例，自动注入
    public ConnectionPoolMonitor(HikariDataSource dataSource) {
        this.dataSource = dataSource;
    }

    // 获取连接池使用率 0.0 ~ 1.0
    public double getConnectionPoolUsage() {
        HikariPoolMXBean bean = dataSource.getHikariPoolMXBean();
        int active = bean.getActiveConnections();
        int max = dataSource.getMaximumPoolSize();
        return (double) active / max;
    }

    // 是否高负载（>80% 表示压力大）
    public boolean isHighLoad() {
        return getConnectionPoolUsage() > 0.8;
    }
}
```

## 三、限流器
```java
@Component
public class MetadataRateLimiter {
    private final RateLimiter rateLimiter = RateLimiter.create(1.0);
    private final DbHealthMetrics metrics = new DbHealthMetrics();

    // 注入连接池监控
    private final ConnectionPoolMonitor poolMonitor;

    public MetadataRateLimiter(ConnectionPoolMonitor poolMonitor) {
        this.poolMonitor = poolMonitor;
    }

    public void acquire() {
        rateLimiter.acquire();//按照设置qps执行，超过限制则阻塞
    }

    public void updateRate(double qps) {
        rateLimiter.setRate(qps);
    }

    // ===================== 【升级】三合一健康判断 =====================
    public boolean needDegrade() {
        // 实时获取连接池使用率
        double poolUsage = poolMonitor.getConnectionPoolUsage();
        metrics.setConnectionPoolUsage(poolUsage);
        
        // DbHealthMetrics增加字段 lastRecordTs
		//if(System.currentTimeMillis() - metrics.getLastRecordTs() > 30000){
		//    // 超过30s没有新样本，重置指标
		//    metrics.reset();
		//}

        // 降级规则（任意满足就降级）
        // 1. 连接池使用率 > 80%（最关键）
        if (poolUsage > 0.8) {
            return true;
        }
        // 2. 平均响应时间 > 200ms
        if (metrics.getAvgRt() > 200) {
            return true;
        }
        // 3. 错误率 > 5%
        if (metrics.getErrorRate() > 5) {
            return true;
        }
        return false;
    }

    // 降级：降低采集速度 + 强制等待
    public void doDegrade() throws InterruptedException {
        updateRate(0.1); // 降到 0.1 QPS
        Thread.sleep(500);
    }

    // 恢复
    public void recover() {
        updateRate(1.0);
    }

    // 记录指标
	public void record(long lastRecordTs ,long rt, boolean isError) {
	    double oldAvgRt = metrics.getAvgRt();
	    double newAvgRt;
	    if (oldAvgRt <= 0) {
	        // 冷启动：第一条样本直接赋值
	        newAvgRt = rt;
	    } else {
	        // EMA 0.9权重历史，0.1权重当前
	        newAvgRt = oldAvgRt * 0.9 + rt * 0.1;
	    }
	    metrics.setAvgRt(newAvgRt);
	
	    metrics.setLastRt(rt);
	    
	    metrics.setLastRecordTs(lastRecordTs);
	
	    // 错误率也使用EMA：error=1，正常=0
	    double errorVal = isError ? 1.0 : 0.0;
	    double oldErrRate = metrics.getErrorRate() / 100.0;
	    double newErrRate;
	    if(oldErrRate < 0){
	        newErrRate = errorVal;
	    }else{
	        newErrRate = oldErrRate * 0.9 + errorVal * 0.1;
	    }
	    metrics.setErrorRate(newErrRate * 100);
	}
}
```

## 四、限流逻辑
```java
private <T> T executeWithProtect(SqlSupplier<T> supplier) throws Exception {
    rateLimiter.acquire();

    if (rateLimiter.needDegrade()) {
        rateLimiter.doDegrade();
    } else {//三项全部不命中,取消限流
        rateLimiter.recover();
    }

    long start = System.currentTimeMillis();
    boolean isError = false;
    try {
        return supplier.get();
    } catch (Exception e) {
        isError = true;
        throw e;
    } finally {
        long rt = System.currentTimeMillis() - start;
        rateLimiter.record(System.currentTimeMillis(),rt, isError);
    }
}
```
