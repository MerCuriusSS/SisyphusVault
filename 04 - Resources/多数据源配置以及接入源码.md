### 1.数据库实例配置
#### 1）实例实体类
```java
import lombok.Data;

/**
 * 数据库实例信息（可以存在 MySQL/PostgreSQL/Redis 中）
 * 支持成百上千条记录
 */
@Data
public class DbInstance {
    private String instanceId;   // 唯一实例ID（如：mysql_001、oracle_002）
    private String dbType;      // 类型：mysql/oracle/pgsql
    private String url;
    private String username;
    private String password;
}
```


#### 2）动态数据源工具类
```java
/**
 * 动态数据源核心类
 * 支持：成百上千个数据库实例 动态连接/查询/销毁
 */
@Component
public class DynamicDbExecutor {
    // 缓存：instanceId -> DataSource（线程安全，支持高并发）
    private final Map<String, DataSource> dataSourceCache = new ConcurrentHashMap<>();

    // ===================== 核心：根据实例ID获取/创建数据源 =====================
    private DataSource getOrCreateDataSource(DbInstance instance) {
        String instanceId = instance.getInstanceId();
        // 缓存存在，直接返回
        if (dataSourceCache.containsKey(instanceId)) {
            return dataSourceCache.get(instanceId);
        }

        // 缓存不存在 → 动态创建连接池
        HikariConfig config = new HikariConfig();
        config.setJdbcUrl(instance.getUrl());
        config.setUsername(instance.getUsername());
        config.setPassword(instance.getPassword());
        config.setDriverClassName(getDriverClass(instance.getDbType()));

        // 关键配置：防止成百上千个连接耗尽资源
        config.setMaximumPoolSize(2);  // 每个实例最多2个连接（按需调）
        config.setMinimumIdle(0);      // 空闲自动释放
        config.setIdleTimeout(60000);  // 空闲1分钟关闭

        // 创建数据源并缓存
        DataSource dataSource = new HikariDataSource(config);
        dataSourceCache.put(instanceId, dataSource);
        return dataSource;
    }

    // ===================== 根据数据库类型获取驱动 =====================
    private String getDriverClass(String dbType) {
        return switch (dbType.toLowerCase()) {
            case "mysql" -> "com.mysql.cj.jdbc.Driver";
            case "oracle" -> "oracle.jdbc.OracleDriver";
            case "pgsql" -> "org.postgresql.Driver";
            default -> throw new RuntimeException("不支持的数据库类型：" + dbType);
        };
    }

    // ===================== 执行查询 SQL =====================
    public Map<String, Object> queryOne(DbInstance instance, String sql) {
        DataSource ds = getOrCreateDataSource(instance);
        return new JdbcTemplate(ds).queryForMap(sql);
    }

    // ===================== 执行列表查询 =====================
    public java.util.List<java.util.Map<String, Object>> queryList(DbInstance instance, String sql) {
        DataSource ds = getOrCreateDataSource(instance);
        return new JdbcTemplate(ds).queryForList(sql);
    }

    // ===================== 执行更新 =====================
    public int update(DbInstance instance, String sql) {
        DataSource ds = getOrCreateDataSource(instance);
        return new JdbcTemplate(ds).update(sql);
    }

    // ===================== 销毁某个实例连接 =====================
    public void removeDataSource(String instanceId) {
        DataSource ds = dataSourceCache.remove(instanceId);
        if (ds instanceof HikariDataSource hds) {
            hds.close();
        }
    }
}
```

### 2.业务层调用示例

```java
// 模拟 1000 个数据库实例（实际从你的数据库中查询）
DbInstance instance1 = new DbInstance();
instance1.setInstanceId("mysql_001");
instance1.setDbType("mysql");
instance1.setUrl("jdbc:mysql://localhost:3306/test");
instance1.setUsername("root");
instance1.setPassword("123456");

DbInstance instance2 = new DbInstance();
instance2.setInstanceId("oracle_001");
instance2.setDbType("oracle");
instance2.setUrl("jdbc:oracle:thin:@ip:1521:ORCL");
instance2.setUsername("system");
instance2.setPassword("123456");


//执行SQL
@Autowired
private DynamicDbExecutor dynamicDbExecutor;

// 查询 MySQL 实例
Map<String, Object> mysqlResult = dynamicDbExecutor.queryOne(instance1, "SELECT * FROM user LIMIT 1");

// 查询 Oracle 实例
List<Map<String, Object>> oracleResult = dynamicDbExecutor.queryList(instance2, "SELECT * FROM user WHERE ROWNUM=1");
```
