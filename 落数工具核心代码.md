```java
// 有界队列：消费者 → 写入器（天然背压）
BlockingQueue<QueueEntry> dataQueue = new LinkedBlockingQueue<>(QUEUE_CAPACITY);

// 启动唯一写入线程
WriterWorker writerWorker = new WriterWorker(dataQueue, fileWriter);
Thread writerThread = new Thread(writerWorker, "writer-thread");

// 启动一批消费者线程
List<KafkaConsumerWorker> consumerWorkers = new ArrayList<>();
List<Thread> consumerThreads = new ArrayList<>();
for (int i = 0; i < consumerThreads; i++) {
    KafkaConsumerWorker worker = new KafkaConsumerWorker(props, dataQueue, i);
    consumerWorkers.add(worker);
    consumerThreads.add(new Thread(worker, "consumer-" + i));
}

// 启动所有线程
writerThread.start();
consumerThreads.forEach(Thread::start);

// JVM 优雅关闭钩子
Runtime.getRuntime().addShutdownHook(new Thread(() -> {
    log.info("Shutting down...");

    // 1. 停止消费者，等待退出
    consumerWorkers.forEach(KafkaConsumerWorker::shutdown);
    consumerThreads.forEach(t -> {
        try { t.join(3000); } catch (InterruptedException ignored) {}
    });

    // 2. 停止写入器，处理完剩余数据
    writerWorker.shutdown();
    try { writerThread.join(5000); } catch (InterruptedException ignored) {}

    log.info("Shutdown complete");
}, "shutdown-hook"));
```

#### 2）消费者任务
```java
// 消费者私有偏移量队列：Writer 写完数据后把待提交的 offset 放入此队列  
private final BlockingQueue<Map<TopicPartition, Long>> ackQueue = new LinkedBlockingQueue<>();

while (running) {
    // 1. 处理ack并提交偏移量
    drainAcksAndCommit();

    // 2. 从Kafka拉取消息
    ConsumerRecords<String, String> records = consumer.poll(Duration.ofSeconds(1));
    if (records.isEmpty()) continue;

    // 3. 消息入队（队列满则阻塞，实现背压）
    for (ConsumerRecord<String, String> record : records) {
        MessageEnvelope envelope = MessageEnvelope.create(
            record.topic(), record.partition(), record.offset(),
            record.key(), record.value()
        );
        try {
            dataQueue.put(new QueueEntry(envelope, consumerId, ackQueue));
        } catch (InterruptedException e) {
            // 恢复中断标志并退出循环
            Thread.currentThread().interrupt();
            running = false;
            break;
        }
    }
}

private void drainAcksAndCommit() {
    List<Map<TopicPartition, Long>> acks = new ArrayList<>();
    ackQueue.drainTo(acks);
    if (acks.isEmpty()) return;

    // 合并：同一分区只保留最大 offset
    Map<TopicPartition, Long> merged = new HashMap<>();
    for (var batch : acks) {
        for (var entry : batch.entrySet()) {
            merged.merge(entry.getKey(), entry.getValue(), Math::max);
        }
    }

    // 封装并同步提交
    Map<TopicPartition, OffsetAndMetadata> toCommit = new HashMap<>();
    merged.forEach((tp, offset) -> toCommit.put(tp, new OffsetAndMetadata(offset)));

    try {
        consumer.commitSync(toCommit);
        log.debug("[{}] Committed offsets: {}", consumerId, merged);
    } catch (Exception e) {
        log.error("[{}] Commit failed, will retry next cycle", consumerId, e);
    }
}
```


#### 3）写入文件任务：
```java
try {
    // 运行中 或 队列未空 → 继续处理
    while (running || !dataQueue.isEmpty()) {
        // 批量拉取最多500条，提升效率
        int drained = dataQueue.drainTo(batch, 500);
        
        if (drained == 0) {
            // 队列为空，阻塞等待新数据
            QueueEntry entry = dataQueue.poll(200, TimeUnit.MILLISECONDS);
            if (entry != null) batch.add(entry);
            continue;
        }

        // 单线程写入文件
        for (QueueEntry entry : batch) {
            fileWriter.write(entry.envelope().toJson());
        }

        // 发送ACK，提交偏移量
        notifyAcks(batch);
        batch.clear();
    }
} catch (InterruptedException e) {
    // 中断恢复标记
    Thread.currentThread().interrupt();
    log.info("WriterWorker interrupted");
} catch (Exception e) {
    log.error("WriterWorker error", e);
} finally {
    // 保证文件关闭
    fileWriter.close();
    log.info("WriterWorker stopped");
}

/**
 * 批量写入完成后，通知对应消费者提交偏移量（写入成功才提交，保证精准一次）
 * 按消费者ackQueue分组，每个分区只保留最大 offset+1
 */
private void notifyAcks(List<QueueEntry> batch) {
    // 按 ackQueue 分组：key=消费者确认队列, value=分区->最大提交偏移量
    Map<BlockingQueue<Map<TopicPartition, Long>>, Map<TopicPartition, Long>> groupByAck = new HashMap<>();

    for (QueueEntry entry : batch) {
        MessageEnvelope env = entry.envelope();
        TopicPartition tp = new TopicPartition(env.topic(), env.partition());
        // Kafka 必须提交 nextOffset = 当前偏移量 + 1
        long nextOffset = env.offset() + 1;

        // 不存在则创建Map，存在则更新为最大偏移量
        groupByAck.computeIfAbsent(entry.ackQueue(), k -> new HashMap<>())
                  .merge(tp, nextOffset, Math::max);
    }

    // 将最终偏移量结果放入对应队列，触发消费者提交
    for (var entry : groupByAck.entrySet()) {
        entry.getKey().offer(entry.getValue());
    }
}
```

### 3.文件大小切割

```java
public void write(String line) {  
    String today = LocalDate.now(ZoneId.systemDefault()).format(FMT);  
  
    // 优先级 1：跨天 → 序列号归零  
    if (!today.equals(currentDate)) {  
        rollToDate(today);  
    }  
  
    // 优先级 2：当前文件继续写会超限 → 序列号递增  
    int estimatedLineBytes = line.length() + LINE_SEP_LEN;  
    if (currentFileBytes + estimatedLineBytes > maxFileSizeBytes) {  
        rollToNextSequence();  
    }  
  
    try {  
        writer.write(line);  
        writer.newLine();  
        writer.flush();  
        currentFileBytes += estimatedLineBytes;  
    } catch (IOException e) {  
        log.error("Failed to write line", e);  
    }  
}

private void rollToDate(String today) {  
    closeWriter();  
    currentDate = today;  
    currentSequence = 0;  
    openCurrentFile();  
    log.info("Rolled to new date: {}", currentDate);  
}  
  
/**  
 * 大小超限切换：序列号 +1，打开新的切割文件。  
 */  
private void rollToNextSequence() {  
    closeWriter();  
    currentSequence++;  
    openCurrentFile();  
    log.info("Rolled to next sequence: {}_{}", currentDate, currentSequence);  
}  
  
/**  
 * 打开当前日期 + 序列号对应的文件。  
 * 若文件已存在则以追加模式打开，并读取当前字节数继续计数（进程重启场景）。  
 */  
private void openCurrentFile() {  
    try {  
        Files.createDirectories(baseDir.resolve(currentDate));  
        Path file = baseDir.resolve(currentDate).resolve(currentDate + "_" + currentSequence + ".txt");  
        writer = Files.newBufferedWriter(file,  
                StandardOpenOption.CREATE, StandardOpenOption.APPEND);  
        currentFileBytes = Files.size(file);  
        log.info("Opened output file: {} ({} bytes existing)", file.toAbsolutePath(), currentFileBytes);  
    } catch (IOException e) {  
        throw new RuntimeException("Failed to open output file", e);  
    }  
}
```

### 4. 缓冲区 提高磁盘IO输入效率
 

```java
/**  
 * 写入一行文本。内部自动判断换日和大小超限并切换文件。  
 * flush 不再是每条执行，而是按时间间隔批量触发，  
 * 将每日上亿次系统调用降到万级，消除 CPU 瓶颈。  
 */  
public void write(String line) {  
    String today = LocalDate.now(ZoneId.systemDefault()).format(FMT);  
  
    // 优先级 1：跨天 → 序列号归零  
    if (!today.equals(currentDate)) {  
        flushIfNeeded(true);  
        rollToDate(today);  
    } else if (currentFileBytes + (line.length() + LINE_SEP_LEN) > maxFileSizeBytes) {  
        // 优先级 2：当前文件继续写会超限 → 序列号递增  
        flushIfNeeded(true);  
        rollToNextSequence();  
    } else {  
        // 优先级 3：正常写入，按时间间隔 flush        flushIfNeeded(false);  
    }  
  
    try {  
        writer.write(line);  
        writer.newLine();  
        currentFileBytes += line.length() + LINE_SEP_LEN;  
    } catch (IOException e) {  
        log.error("Failed to write line", e);  
    }  
}


/**  
 * 按时间间隔触发 flush。  
 * @param force true 时强制 flush（换日、切文件时使用）  
 */  
private void flushIfNeeded(boolean force) {  
    if (writer == null) {  
        return;  
    }  
    long now = System.nanoTime();  
    if (force || (now - lastFlushTime) >= flushIntervalNanos) {  
        try {  
            writer.flush();  
            lastFlushTime = now;  
        } catch (IOException e) {  
            log.error("Failed to flush writer", e);  
        }  
    }  
}
```

```java
    /**
     * 扫描当天目录所有txt文件，返回各分区最大落盘offset
     */
    public static Map<TopicPartition, Long> recoverOffsets(Path outputDir) {
        // 构建当天日期目录
        String today = LocalDate.now(ZoneId.systemDefault()).format(FMT);
        Path dateDir = outputDir.resolve(today);
        
        if (!Files.isDirectory(dateDir)) {
            return Collections.emptyMap();
        }

        Map<TopicPartition, Long> result = new HashMap<>();

        // 遍历所有txt文件
        try (DirectoryStream<Path> files = Files.newDirectoryStream(dateDir, "*.txt")) {
            for (Path file : files) {
                // 合并每个文件的最大offset
                scanFileTail(file).forEach((tp, offset) ->
                    result.merge(tp, offset, Math::max)
                );
            }
        } catch (Exception e) {
            return Collections.emptyMap();
        }

        return result;
    }
```

```java
    /**
     * 读取文件尾部，解析有效JSON行获取分区最大offset
     */
    private static Map<TopicPartition, Long> scanFileTail(Path file) {
        Map<TopicPartition, Long> offsets = new HashMap<>();

        try (RandomAccessFile raf = new RandomAccessFile(file.toFile(), "r")) {
            long fileSize = raf.length();
            if (fileSize == 0) return offsets;

            // 读取文件尾部指定字节
            long start = Math.max(0, fileSize - TAIL_BYTES);
            byte[] tail = new byte[(int) (fileSize - start)];
            raf.seek(start);
            raf.readFully(tail);

            // 分割行并跳过可能截断的首行
            String[] lines = new String(tail, StandardCharsets.UTF_8).split("\r?\n");
            int firstLine = start > 0 ? 1 : 0;

            for (int i = firstLine; i < lines.length; i++) {
                String line = lines[i].trim();
                if (line.isEmpty()) continue;

                try {
                    // 解析JSON获取分区和offset
                    JsonNode node = MAPPER.readTree(line);
                    TopicPartition tp = new TopicPartition(
                        node.get("topic").asText(),
                        node.get("partition").asInt()
                    );
                    long offset = node.get("offset").asLong();
                    offsets.merge(tp, offset, Math::max);
                } catch (Exception ignored) {
                    // 跳过不完整/损坏行
                }
            }
        } catch (Exception ignored) {}

        return offsets;
    }
```

```java
public void run() {
	// 启动阶段：等待分区分配 → 按文件恢复的 offset 跳位  
	seekPastRecoveredOffsets();
	//从kafka拉取数据...（consumer.poll(Duration.ofMillis(1000));）
}

/**
 * 首次启动：空poll触发分区分配，将每个分区seek到【已落盘offset + 1】，避免重复消费
 */
private void seekPastRecoveredOffsets() {
    if (recoveredOffsets.isEmpty()) return;

    // 1.通过空poll触发分区分配（最多5秒）
    int retries = 0;
    while (consumer.assignment().isEmpty() && retries < 50) {
        consumer.poll(Duration.ofMillis(100));
        retries++;
    }

	//2.获取消费者所分配的分区
    Set<TopicPartition> assignment = consumer.assignment();
    if (assignment.isEmpty()) return;

    // 3.对已分配分区执行seek
    for (TopicPartition tp : assignment) {
        Long offset = recoveredOffsets.get(tp);
        if (offset != null) {
            consumer.seek(tp, offset + 1);
        }
    }
}
```
